% 2.3 SECOND-PRICE AUCTION
% ==============================================================

\begin{frame}[standout]
Second-Price Auctions
\end{frame}

\begin{frame}{The Rules}
A \textbf{second-price} (or \textbf{Vickrey}) auction:
\begin{itemize}
    \item sealed bids: nobody sees anyone else's bid
    \item highest bidder wins
    \item the winner pays the \textbf{second-highest} bid, not their own
    \item ties are broken by a rule fixed in advance
\end{itemize}
\bsk
Why would a seller ever do this? It looks like giving money away.
\vfill
\vspace{0.9in}
\end{frame}

\begin{frame}{An Example: The Game-Day Permit}
One food truck permit for the lot outside the stadium is up for auction.
Dos Hermanos and Pitabilities both want it.
\bsk
\begin{itemize}
    \item Dos Hermanos has run the numbers: the permit is worth \textbf{\$30k} to them
    \item Bids must be one of \$10k, \$20k, \$30k, \$40k, \$50k
    \item Ties go to Pitabilities (decided by a coin flip beforehand)
\end{itemize}
\bsk
Dos Hermanos's payoff:
$$\pi_1 = \begin{cases} 30 - (\text{Pitabilities' bid}) & \text{if Dos Hermanos wins} \\ 0 & \text{otherwise}\end{cases}$$
\vfill
Note they pay \textit{Pitabilities'} bid, not their own.
\vspace{-0.4in}
\end{frame}

\begin{frame}{Only Dos Hermanos's Payoffs Shown}
\centering
{\small
\begin{matrixgame}[top={10, 20, 30, 40, 50}, left={10, 20, 30, 40, 50}, player names={Dos Hermanos (value 30), Pitabilities}]
  \payoffmatrix{0,\cdot & 0,\cdot & 0,\cdot & 0,\cdot & 0,\cdot \\
                20,\cdot & 0,\cdot & 0,\cdot & 0,\cdot & 0,\cdot \\
                20,\cdot & 10,\cdot & 0,\cdot & 0,\cdot & 0,\cdot \\
                20,\cdot & 10,\cdot & 0,\cdot & 0,\cdot & 0,\cdot \\
                20,\cdot & 10,\cdot & 0,\cdot & -10,\cdot & 0,\cdot}
\end{matrixgame}}
\bsk
\raggedright
Check one cell: Dos Hermanos bids 30, Pitabilities bids 10. \vspace{0.5in}
\end{frame}

\begin{frame}{Reading the Table}
\begin{itemize}
    \item Bidding \textbf{30} (their true value) is weakly dominant \vspace{0.3in}
    \item Bidding \textbf{40} is equivalent to bidding 30, also weakly dominant \vspace{0.3in}
    \item Bidding \textbf{50} is strictly worse than 30. Where, and why? \vspace{0.3in}
    \item Bidding \textbf{20} or \textbf{10} is weakly worse. Where, and why? \vspace{0.3in}
\end{itemize}
\end{frame}

\begin{frame}{Vickrey's Theorem}
\textbf{Theorem (Vickrey, 1961).} In a second-price auction, if player $i$ is selfish and
greedy, then bidding their true value ($b_i = v_i$) is a \textbf{weakly dominant strategy}.
\vfill
Remarkable, because:
\begin{itemize}
    \item it holds no matter how many bidders there are
    \item it holds no matter what anyone else bids
    \item you don't need to guess, predict, or model your rivals at all
\end{itemize}
\vfill
Truth-telling isn't nice here. It's just optimal.
\vspace{-0.4in}
\end{frame}

\begin{frame}{Why It Works: Two Cases}
Let $v_1$ be my value and $b_2$ the other bid. Compare bidding $v_1$ to anything else.
\bsk
\textbf{Case 1: $b_2\le v_1$.} \ Bidding $v_1$ wins and pays $b_2$, for a payoff of $v_1-b_2\ge 0$.
\begin{itemize}
    \item Bidding higher: still win, still pay $b_2$. \underline{\hspace{1.5in}}
    \item Bidding below $b_2$: lose, payoff 0. \underline{\hspace{1.5in}}
\end{itemize}
\bsk
\textbf{Case 2: $b_2>v_1$.} \ Bidding $v_1$ loses, for a payoff of 0.
\begin{itemize}
    \item Bidding anything below $b_2$: still lose. \underline{\hspace{1.5in}}
    \item Bidding at or above $b_2$: win, pay $b_2$. \underline{\hspace{1.5in}}
\end{itemize}
\bsk
Raising your bid can only ever change \textit{whether} you win, never \textit{what you pay}.
\end{frame}

\begin{frame}{The Fine Print}
The theorem assumes the bidder is \textbf{selfish and greedy}.
\bsk
Drop that and it fails. Suppose I also care what \textit{you} pay:
\begin{itemize}
    \item \textbf{Spiteful}: if I lose, I'd rather you paid more \vspace{0.35in}
    \item \textbf{Benevolent}: if I lose, I'd rather you paid less \vspace{0.35in}
\end{itemize}
\bsk
In either case, bidding my value is no longer dominant.
\vfill
Same mechanism, different preferences, different game. Again.
\vspace{-0.3in}
\end{frame}

\begin{frame}{Back to Your Bids}
Under second-price, truthful bidding is weakly dominant.
\bsk
Under first-price, it is not\ldots\ winning means paying your own bid, so bidding your
value guarantees a payoff of exactly zero. You must \textbf{shade} down.
\vfill
\centering
\begin{tabular}{lc}
Round 1 (first-price), average bid/value: & \underline{\hspace{1in}} \\[0.3in]
Round 2 (second-price), average bid/value: & \underline{\hspace{1in}} \\[0.3in]
\end{tabular}
\vfill
\end{frame}

% 2.4 PIVOTAL MECHANISM
% ==============================================================

\begin{frame}[standout]
The Pivotal Mechanism
\end{frame}

\begin{frame}{A Real Headline}
\textit{Davis Enterprise}, January 12, 2001:
\bsk
\begin{quote}
``By consensus, the Davis City Council agreed Wednesday to order a community-wide
public opinion poll to gauge how much Davis residents would be willing to pay for a
park tax and a public safety tax.''
\end{quote}
\bsk
Will people answer honestly?
\begin{itemize}
    \item If my answer doesn't affect my tax bill: \vspace{0.35in}
    \item If my answer does affect my tax bill: \vspace{0.35in}
\end{itemize}
\end{frame}

\begin{frame}{The Setup}
A public project such as building the Upper Arlington Community Center costs
\$$C$ in total. There are $n$ residents.
\bsk
\begin{itemize}
    \item Resident $i$'s share of the cost is $c_i$, with $c_1+\cdots+c_n=C$ \\
      {\footnotesize (shares are announced in advance and need not be equal)}
    \item Resident $i$ starts with wealth $m_i$
    \item If built, resident $i$ gets benefit worth $v_i$ (possibly \textbf{negative}) \\
      {\footnotesize (you live on the street it's going on: traffic, lights, parking)}
\end{itemize}
\bsk
$$U_i(\$m)=\begin{cases} m & \text{if the project is not built}\\ m+v_i & \text{if it is}\end{cases}$$
\vfill
\textbf{Efficiency:} build it exactly when the project is worth more to the residents
together than it costs to build. Social benefit beats total cost.
\bsk
$$\underbrace{\textstyle\sum_{i=1}^n v_i}_{\text{social benefit}}\ \ >\ \ \underbrace{C\vphantom{\textstyle\sum}}_{\text{total cost}}$$
\vspace{-0.4in}
\end{frame}

\begin{frame}{Why Efficiency Isn't Enough}
$n=2$, \ $m_1=50$, \ $m_2=60$, \ $v_1=19$, \ $v_2=-15$, \ $C=6$, \ $c_1=6$, \ $c_2=0$.
\bsk
$\sum v_i = 19-15 = 4 < 6 = C$, so it should \textit{not} be built.
\bsk
\centering
\begin{tabular}{l|c|c|}
\cline{2-3}
 & Not built & Built \\
\hline
\multicolumn{1}{|l|}{Resident 1} & $50$ & $50+19-6=63$ \\
\hline
\multicolumn{1}{|l|}{Resident 2} & $60$ & $60-15=45$ \\
\hline
\end{tabular}
\bsk
\raggedright
Gain of 13 to one, loss of 15 to the other.
\vfill
The government would happily follow the rule if it knew $v_1$ and $v_2$.
\bsk
It doesn't. And asking is exactly the problem.
\vspace{-0.3in}
\end{frame}

\begin{frame}{Clarke's Idea}
Each resident $i$ reports a number $w_i$, which represents their reported benefit (or harm, if negative).\\
They may lie: nothing forces $w_i=v_i$.
\bsk
\textbf{Decision rule:}
$$\text{Build?}\quad\begin{cases} \text{Yes} & \text{if } \sum_{j=1}^n w_j > C\\[2pt]
\text{No} & \text{if } \sum_{j=1}^n w_j \le C\end{cases}$$
\bsk
\textbf{Then the twist.} Resident $i$ is \textbf{pivotal} if the decision \textit{without their report}
would differ from the decision \textit{with} it.
\begin{itemize}
    \item Not pivotal $\Rightarrow$ they pay no tax
    \item Pivotal $\Rightarrow$ they pay $\Big|\sum_{j\ne i}w_j-\sum_{j\ne i}c_j\Big|$
\end{itemize}
\end{frame}

\begin{frame}{Worked Example}
$n=3$, \ $C=10$, \ $c_1=3$, $c_2=2$, $c_3=5$. \ Reports: $w_1=-1$, $w_2=8$, $w_3=3$.
\bsk
$\sum w_j=10$, which is \textit{not} greater than $C=10$, so the project is \textbf{not built}.
\bsk
\centering
{\small
\begin{tabular}{|c|c|c|c|c|}
\hline
$i$ & $\sum_{j\ne i}w_j$ & $\sum_{j\ne i}c_j$ & Decision without $i$ & Tax \\
\hline
$1$ & $8+3=11$ & $2+5=7$ & Build (pivotal) & $|11-7|=4$ \\
\hline
$2$ & $-1+3=2$ & $3+5=8$ & Don't (not pivotal) & $0$ \\
\hline
$3$ & $-1+8=7$ & $3+2=5$ & Build (pivotal) & $|7-5|=2$ \\
\hline
\end{tabular}}
\bsk
\raggedright
\vfill
Note residents 1 and 3 each flipped the outcome, and each pays for it.
\vspace{-0.3in}
\end{frame}

\begin{frame}{``Then I'll Just Avoid Being Pivotal''}
You can guarantee it: report $w_i=c_i$ and you are never pivotal, never taxed.
\bsk
Is that smart? Take $n=4$, $C=15$, $c=(5,0,5,5)$. You are resident 1, with
$m_1=40$ and $v_1=25$. You expect $w_2=-40$, $w_3=15$, $w_4=20$.
\bsk
\textbf{Report $w_1=c_1=5$:} \ $\sum w_j = 0 < 15$, not built, and you pay nothing.
$$U_1 = 40$$
\bsk
\textbf{Report truthfully, $w_1=25$:} \ $\sum w_j = 20 > 15$, so it \textit{is} built.
You are pivotal, taxed $|(-5)-(10)|=15$.
$$U_1 = 40 + 25 - 5 - 15 = 45$$
\vspace{-0.3in}
\end{frame}

\begin{frame}{Clarke's Theorem}
\vfill
\centering
\textbf{Theorem (Clarke, 1971).} In the pivotal mechanism, reporting truthfully
($w_i=v_i$) is a \textbf{weakly dominant strategy} for every player.
\vfill
\raggedright
Same shape as Vickrey: the tax you pay is built out of \textit{other people's} numbers,
so your report moves the decision without moving your own bill.
\vfill
\vspace{0.5in}
\end{frame}

% ==============================================================
% 2.5 ITERATED DELETION
% ==============================================================

\begin{frame}[standout]
Iterated Deletion
\end{frame}

\begin{frame}{The Problem}
If a player has a dominant strategy, they should play it.
\bsk
But most games don't have one. So what then?
\vfill
New idea: even without a dominant strategy, a rational player will never play a
\textbf{strictly dominated} one.
\bsk
And if I know you're rational, I know you won't either\ldots\ which shrinks the game
I'm actually facing.
\bsk
And then \textit{that} smaller game may have dominated strategies of its own.
\vspace{-0.4in}
\end{frame}

\begin{frame}{IDSDS}
\textbf{Iterated Deletion of Strictly Dominated Strategies.}
\bsk
Start with game $G$.
\begin{itemize}
    \item $G^1$: delete every strictly dominated strategy, for every player
    \item $G^2$: do it again, inside $G^1$
    \item \ \ $\vdots$
    \item $G^\infty$: the output, reached in finitely many steps if $G$ is finite
\end{itemize}
\bsk
If $G^\infty$ is a single strategy profile, we call it the
\textbf{iterated strict dominant-strategy profile}.\\
Otherwise it's just ``the output of IDSDS.''
\vfill
{\footnotesize Order doesn't matter: any sequence of deletions of strictly dominated
strategies reaches the same output.}
\end{frame}

\begin{frame}{Doing It}
\centering
\begin{matrixgame}[top={L, C, R}, left={T, M, B}, player names={Player 1, Player 2}]
  \payoffmatrix{4,3 & 5,1 & 6,2 \\ 2,1 & 8,4 & 3,6 \\ 3,0 & 9,6 & 2,8}
\end{matrixgame}
\bsk
\raggedright
Nobody has a dominant strategy here. Check a couple:
\begin{itemize}
    \item $T$ vs.\ $M$ for Player 1: \ $4>2$, but $5<8$ \ $\Rightarrow$ neither dominates
    \item $T$ vs.\ $B$ for Player 1: \ $6>2$, but $5<9$ \ $\Rightarrow$ neither dominates
\end{itemize}
\bsk
So start with Player 2. \vspace{0.5in}
\end{frame}

\begin{frame}{Doing It}
\textbf{Step 1.} For Player 2, $R$ strictly dominates $C$: \ $2>1$, \ $6>4$, \ $8>6$.
Delete $C$.
\bsk
\centering
\begin{matrixgame}[top={L, R}, left={T, M, B}, player names={Player 1, Player 2}]
  \payoffmatrix{4,3 & 6,2 \\ 2,1 & 3,6 \\ 3,0 & 2,8}
\end{matrixgame}
\bsk
\raggedright
\textbf{Step 2.} Now $T$ strictly dominates \textit{both} $M$ and $B$ for Player 1,
which it did not do a moment ago. Delete them.
\vspace{-0.2in}
\end{frame}

\begin{frame}{Doing It}
\centering
\begin{matrixgame}[top={L, R}, left={T}, player names={Player 1, Player 2}]
  \payoffmatrix{4,3 & 6,2}
\end{matrixgame}
\bsk
\raggedright
\textbf{Step 3.} Player 2 now faces a one-row game: $L$ gives 3, $R$ gives 2.
Delete $R$.
\vfill
\centering
\large
$G^\infty=\{(T,L)\}$
\bsk
\normalsize
\raggedright
Nobody had a dominant strategy at the start, and yet the game has a unique
prediction, \textit{provided} each player knows the other is rational, and knows
that the other knows, and so on.
\vspace{-0.3in}
\end{frame}

\begin{frame}{Bonanno's Example}
Four rounds instead of three.
\bsk
\centering
\begin{matrixgame}[top={e, f, g, h}, left={A, B, C, D}, player names={Player 1, Player 2}]
  \payoffmatrix{6,3 & 4,4 & 4,1 & 3,0 \\ 5,4 & 6,3 & 0,2 & 5,1 \\ 5,0 & 3,2 & 6,1 & 4,0 \\ 2,0 & 2,3 & 3,3 & 6,1}
\end{matrixgame}
\bsk
\raggedright
\begin{enumerate}
    \item delete $h$ \ (dominated by $g$) \hspace{0.4in} 3. delete $g$ \ (dominated by $f$)
    \item delete $D$ \ (dominated by $C$) \hspace{0.35in} 4. delete $C$ \ (dominated by $A$)
\end{enumerate}
\bsk
Output: $\{(A,e),(A,f),(B,e),(B,f)\}$, not a single profile.
\vspace{-0.3in}
\end{frame}

\begin{frame}{Back to Your Numbers}
$\tfrac{2}{3}$ of the average can never exceed $\tfrac{2}{3}\times 100\approx 66.7$.
\begin{itemize}
    \item So $67$ strictly dominates everything above it. Delete $68$--$100$. \vspace{0.25in}
    \item Now the target can't exceed $\tfrac{2}{3}\times 67\approx 44.7$. Delete $45$--$67$. \vspace{0.25in}
    \item Now it can't exceed $\tfrac{2}{3}\times 45=30$. Delete $31$--$45$. \vspace{0.25in}
    \item Keep going \ldots \ where does it end? \vspace{0.25in}
\end{itemize}
\vfill
So why didn't we all play that in round 1?
\vspace{-0.3in}

\begin{frame}{IDWDS, and Why It's Trickier}
What if we delete \textbf{weakly} dominated strategies too?
\bsk
\centering
\begin{matrixgame}[top={L, R}, left={A, T, M, B}, player names={Player 1, Player 2}]
  \payoffmatrix{4,0 & 0,0 \\ 3,2 & 2,2 \\ 1,1 & 0,0 \\ 0,0 & 1,1}
\end{matrixgame}
\bsk
\raggedright
Two perfectly reasonable places to start:
\begin{itemize}
    \item $M$ is strictly dominated by $T$ \ \ \ \ \ ($3>1$, $2>0$)
    \item $B$ is strictly dominated by $T$ \ \ \ \ \ ($3>0$, $2>1$)
\end{itemize}
\vspace{-0.2in}
\end{frame}

\begin{frame}{Order One: Delete $M$ First}
\centering
Delete $M$ \ $\Rightarrow$ \ for Player 2, $R$ now weakly dominates $L$ \ $\Rightarrow$ delete $L$
\bsk
\begin{matrixgame}[top={R}, left={A, T, B}, player names={Player 1, Player 2}]
  \payoffmatrix{0,0 \\ 2,2 \\ 1,1}
\end{matrixgame}
\bsk
\raggedright
Now $A$ and $B$ are strictly dominated by $T$.
\vfill
\centering
\large
Answer: \ $(T,R)$, \ payoffs $(2,2)$
\vfill
\end{frame}

\begin{frame}{Order Two: Delete $B$ First}
\centering
Delete $B$ \ $\Rightarrow$ \ for Player 2, $L$ now weakly dominates $R$ \ $\Rightarrow$ delete $R$
\bsk
\begin{matrixgame}[top={L}, left={A, T, M}, player names={Player 1, Player 2}]
  \payoffmatrix{4,0 \\ 3,2 \\ 1,1}
\end{matrixgame}
\bsk
\raggedright
Now $T$ and $M$ are strictly dominated by $A$.
\vfill
\centering
\large
Answer: \ $(A,L)$, \ payoffs $(4,0)$
\vfill
\end{frame}

\begin{frame}{So the Procedure Isn't Well Defined}
Two legitimate orders, two different answers\ldots\ and not by a little:
$(2,2)$ versus $(4,0)$.
\vfill
\textbf{Fix (Definition).} At every step, identify \textit{all} weakly or strictly dominated
strategies for \textit{all} players, and delete them \textbf{simultaneously}.
\bsk
Applied to that game, IDWDS gives $\{(A,L),(A,R),(T,L),(T,R)\}$.
\vfill
Note this is not a single profile, so the game has no iterated weak
dominant-strategy solution.
\vspace{-0.4in}
\end{frame}

\begin{frame}{A Word of Caution}
IDSDS has a clean story: rationality, plus knowing others are rational, and so on.
\bsk
IDWDS does not.
\begin{itemize}
    \item To delete a \textit{weakly} dominated strategy, you must believe your opponent
      might play anything. Otherwise the tie-breaking case never arises
    \item But deletion is precisely the act of ruling strategies out
\end{itemize}
\vfill
So the justification needs rationality \textit{and} a notion of caution, and the two
pull against each other.
\bsk
We won't resolve that here. Just don't treat the two procedures as equally innocent.
\vspace{-0.3in}
\end{frame}

% 2.6 NASH EQUILIBRIUM
% ==============================================================

\begin{frame}[standout]
Nash Equilibrium
\end{frame}

\begin{frame}{Still Not Enough}
Games solved outright by IDSDS or IDWDS are the exception.
\bsk
We need something that applies to \textit{every} game.
\vfill
The idea: stop asking ``what will a rational player do?'' and start asking
\bsk
\centering
\large
``which profiles are \textbf{stable}?''
\vfill
\end{frame}

\begin{frame}{Definition: Nash Equilibrium (Two Players)}
A strategy profile $s^*=(s_1^*,s_2^*)$ is a \textbf{Nash equilibrium} if
\bsk
\begin{enumerate}
    \item for every $s_1\in S_1$: \hspace{0.3in} $\pi_1(s_1^*,s_2^*)\ \ge\ \pi_1(s_1,s_2^*)$
    \bsk
    \item for every $s_2\in S_2$: \hspace{0.3in} $\pi_2(s_1^*,s_2^*)\ \ge\ \pi_2(s_1^*,s_2)$
\end{enumerate}
\vfill
In words: \textbf{holding the other player's strategy fixed}, neither player can do
better by changing their own.
\bsk
Note what's fixed and what varies in each line. That's the whole definition.
\vspace{-0.4in}
\end{frame}

\begin{frame}{Four Ways to Say It}
\textbf{No regret.} After seeing what the other player did, nobody wishes they had
chosen differently.
\bsk
\textbf{Self-enforcing agreement.} If the players could talk beforehand and shake on
$s^*$, nobody would want to break it, assuming the other keeps it.
\bsk
\textbf{Viable recommendation.} If a third party publicly recommended $s^*$, nobody
would want to deviate, assuming the others comply.
\bsk
\textbf{Rationality with correct beliefs.} Each player best-responds to what they
believe the other will do, and those beliefs are right.
\vfill
{\footnotesize These are all the same inequalities in different clothes.}
\vspace{-0.3in}
\end{frame}

\begin{frame}{Best Replies}
Strategy $s_i$ is a \textbf{best reply} to $s_{-i}$ if
$$\pi_i(s_i,s_{-i})\ \ge\ \pi_i(s_i',s_{-i})\quad\text{for every } s_i'\in S_i$$
\bsk
Equivalent definition of Nash equilibrium:
\vfill
\centering
\large
$s$ is a Nash equilibrium \ $\iff$ \ every $s_i$ is a best reply to $s_{-i}$
\vfill
\normalsize
\raggedright
This is the version worth memorizing. It also tells you how to \textit{find} equilibria.
\vspace{-0.4in}
\end{frame}

    \item In each \textbf{row}, underline Player 2's largest payoff (all ties)

\end{enumerate}

\end{frame}

\begin{frame}{More Than Two Players}
A profile $s^*\in S$ is a Nash equilibrium if, for every player $i$,
$$\pi_i(s^*)\ \ge\ \pi_i(s_1^*,\ldots,s_{i-1}^*,\ s_i,\ s_{i+1}^*,\ldots,s_n^*)
\quad\text{for all } s_i\in S_i$$
\bsk
$n$ inequalities, one per player. Nothing new, just more of them.
\vfill
\vspace{0.8in}
\end{frame}

\begin{frame}{Nash Survives IDSDS}
All three procedures return a \textbf{set of strategy profiles}, not one prediction:
\begin{itemize}
    \item $NE(G)$: the profiles satisfying the Nash inequalities
    \item $IDSDS(G)$: the profiles still standing when strict deletion halts
    \item $IDWDS(G)$: the same, for weak deletion
\end{itemize}
Each is a subset of $S$, and each can be empty, a single profile, or many.
\vfill
\textbf{Theorem.} \ For every ordinal strategic-form game $G$,
$$NE(G)\ \subseteq\ IDSDS(G)$$
\begin{itemize}
    \item A Nash equilibrium never uses a strategy that IDSDS deletes
    \item So IDSDS never throws away an equilibrium: it's safe
    \item But the same is \textbf{not} true of IDWDS: it is possible that
      $NE(G)\ne\emptyset$ and yet $NE(G)\cap IDWDS(G)=\emptyset$
\end{itemize}
\vspace{-0.3in}
\end{frame}

\begin{frame}{Strict Nash Equilibrium}
$s^*$ is a \textbf{strict} Nash equilibrium if every player is \textit{strictly} worse off
deviating, rather than merely no better off.
\bsk
\centering
\begin{tabular}{ll}
Nash: & $\pi_i(s^*)\ \ge\ \pi_i(s_i,s_{-i}^*)$ \ \ for every $s_i\in S_i$ \\[0.18in]
Strict: & $\pi_i(s^*)\ >\ \pi_i(s_i,s_{-i}^*)$ \ \ for every $s_i\ne s_i^*$ \\
\end{tabular}
\bsk
\raggedright
Two changes: $\ge$ becomes $>$, and $s_i^*$ itself is excluded. It has to be, since
$\pi_i(s^*)>\pi_i(s^*)$ is false.
\vfill
\textbf{Theorem.} \ If $IDSDS(G)=\{s^*\}$ then $s^*$ is a strict Nash equilibrium.
\bsk
If $IDWDS(G)=\{s^*\}$ then $s^*$ is a Nash equilibrium, but not necessarily a strict one.
\vspace{-0.3in}
\end{frame}

\begin{frame}{A Bigger Example}
Suppose OSU donor Ratmir Timashev makes the following offer to 50 students.
Each secretly writes down a request: a multiple of \$10, up to \$100.
\bsk
\begin{itemize}
    \item If \textbf{5 or fewer} students ask for \$100, everyone gets what they asked for
    \item Otherwise, everyone gets nothing
\end{itemize}
\bsk
All students are selfish and greedy. What are the Nash equilibria?
\vfill
\vspace{1.1in}
\end{frame}

\begin{frame}{A Bigger Example}
There are two families of equilibria.
\bsk
\textbf{(a) Seven or more students request \$100.}\\
Everyone gets nothing. Can anyone gain by switching alone? \vspace{0.4in}
\bsk
\textbf{(b) Exactly five request \$100, and all 45 others request \$90.}\\
Everyone is paid. Can anyone gain by switching alone? \vspace{0.4in}
\bsk
{\footnotesize Convince yourself that \textit{exactly six} requesting \$100 is not an equilibrium.}
\end{frame}

\begin{frame}{When There Is None}
Matching Pennies. Both match $\Rightarrow$ Player 1 wins; otherwise Player 2 wins.
\bsk
\centering
\begin{matrixgame}[top={H, T}, left={H, T}, player names={Player 1, Player 2}, br]
  \payoffmatrix{1,0 & 0,1 \\ 0,1 & 1,0}
\end{matrixgame}
\bsk
\raggedright
No cell has both payoffs underlined.
\vfill
With only ordinal payoffs, a game need not have \textbf{any} Nash equilibrium.
\bsk
Fixing this requires letting players randomize, and randomizing requires
cardinal payoffs. That's Chapter 5.
\vspace{-0.3in}
\end{frame}

\begin{frame}[standout]
Infinite Strategy Sets
\end{frame}

\begin{frame}{No Table, Same Definition}
Two players. Each writes down a real number $\ge 1$, so $S_1=S_2=[1,\infty)$.
\bsk
Player 1 writes $x$, Player 2 writes $y$:
$$\pi_1(x,y)=\begin{cases} x-1 & \text{if } x<y\\ 0 & \text{if } x\ge y\end{cases}
\qquad
\pi_2(x,y)=\begin{cases} y-1 & \text{if } x>y\\ 0 & \text{if } x\le y\end{cases}$$
\bsk
You want to write the larger\ldots\ no, the \textit{smaller} number. But not too small.
\vfill
What are the Nash equilibria? \vspace{0.7in}
\end{frame}

\begin{frame}{The Answer Is $(1,1)$, and Only $(1,1)$}
\textbf{It is an equilibrium.} At $(1,1)$ both get 0. If Player 1 switches to any
$x>1$, then $x\ge y=1$, so they still get 0.
\bsk
\textbf{Nothing else is.}
\begin{itemize}
    \item If $x=y>1$: Player 1 gets 0, but any $\hat x$ with $1<\hat x<x$ gives
      $\hat x-1>0$
    \item If $x<y$: Player 1 gets $x-1$, but any $\hat x$ with $x<\hat x<y$ gives
      $\hat x-1>x-1$
    \item If $y<x$: same argument for Player 2
\end{itemize}
\vfill
\textbf{And now the strange part.} For Player 1, $x=1$ is \textbf{weakly dominated}
by every other strategy.
\bsk
So the unique Nash equilibrium has both players using a weakly dominated strategy.
\vspace{-0.3in}
\end{frame}

\begin{frame}{Cournot Competition}
{\footnotesize (This one uses calculus.)}
\bsk
Shell and Speedway sell an identical good: a gallon of regular. Station $i$
chooses quantity $q_i\ge 0$ at constant unit cost $c$. The price is a \textit{function}
of total output:
$$P(q_1,q_2)=a-b(q_1+q_2),\qquad a>c,\ b>0$$
\bsk
Shell's profit:
$$\pi_1(q_1,q_2)=P(q_1,q_2)\,q_1-cq_1=(a-c)q_1-b q_1^2-b q_1 q_2$$
\bsk
Strategy sets are $[0,\infty)$ (infinite). But a Nash equilibrium is still just a
pair $(\bar q_1,\bar q_2)$ where neither station wants to move.
\vfill
{\footnotesize Cournot wrote this down in 1838, inventing Nash equilibrium about
120 years early, for this one class of games.}
\vspace{-0.3in}
\end{frame}

\begin{frame}{Variables and Parameters}
Not every letter in a model plays the same role.
\bsk
\textbf{Parameters:} \ $a$, $b$, $c$
\begin{itemize}
    \item fixed numbers, known to everyone, chosen by nobody
    \item placeholders for a number that is already settled before anyone moves
    \item left as letters so one solution covers every version of the game
\end{itemize}
\bsk
\textbf{Choice variables:} \ $q_1$, $q_2$
\begin{itemize}
    \item placeholders for a number that someone gets to pick
    \item these are the strategies: $S_i=[0,\infty)$
\end{itemize}
\bsk
\textbf{Derived values} (functions of the variables): \ $P(q_1,q_2)$, \ $\pi_i(q_1,q_2)$
\begin{itemize}
    \item not fixed in advance, and nobody chooses them directly
    \item they are \textit{functions}: feed in the choices, get a number out
\end{itemize}
\vfill
For any letter, ask: could a player have made it come out differently? \vspace{0.3in}
\end{frame}

% ---- Calculus reminder: partials, FOC, SOC --------------------------
\begin{frame}{Maximizing, With Two Variables}
From the math review: to maximize $f(x)$, set $f'(x)=0$.
\bsk
Shell's profit depends on \textit{two} quantities:
$$\pi_1(q_1,q_2)=(a-c)q_1-bq_1^2-bq_1q_2$$
\bsk
But Shell only chooses $q_1$. Speedway's $q_2$ is not Shell's to pick.
\vfill
So treat $q_2$ as a fixed number and maximize over $q_1$ alone. \vspace{0.8in}
\end{frame}

\begin{frame}{Partial Derivatives}
A \textbf{partial derivative} is an ordinary derivative taken while every other variable
is held constant, treated as though it were just a number.
\bsk
$$\frac{\partial f}{\partial x}\ :\quad \text{differentiate in } x, \text{ treat } y
\text{ as a constant}$$
\bsk
The curly $\partial$ is the only new thing. The rules are the ones you already know.
\vfill
$f(x,y)=3x^2+5xy+y^3$
\bsk
$\dfrac{\partial f}{\partial x}=$ \underline{\hspace{2in}} \vspace{0.4in}

$\dfrac{\partial f}{\partial y}=$ \underline{\hspace{2in}} \vspace{0.3in}
\end{frame}

\begin{frame}{Why That Works}
Same $f(x,y)=3x^2+5xy+y^3$. Suppose $y$ is stuck at $2$:
$$f(x,2)=3x^2+10x+8$$
\bsk
That is a function of $x$ alone. Differentiate it the ordinary way:
\bsk
$\dfrac{d}{dx}f(x,2)=$ \underline{\hspace{2in}} \vspace{0.5in}

Now compare it to $\dfrac{\partial f}{\partial x}$ with $y=2$ substituted in.
\vfill
Holding a variable constant and plugging in a constant are the same operation.
\vspace{0.5in}
\end{frame}

\begin{frame}{Finding the Maximizer}
Two conditions, in words first:
\begin{itemize}
    \item \textbf{First order}: at the top, the slope in the $x$ direction is flat
    \item \textbf{Second order}: the curve bends downward there, so flat means a peak
      rather than a valley
\end{itemize}
\bsk
$$\underbrace{\frac{\partial f}{\partial x}=0}_{\text{slope is flat}}
\qquad\text{and}\qquad
\underbrace{\frac{\partial^2 f}{\partial x^2}<0}_{\text{bending downward}}$$
\vfill
Skip the second condition and a flat point could just as easily be a minimum.
\vspace{0.7in}
\end{frame}

\begin{frame}{Shell's Problem}
$$\pi_1(q_1,q_2)=(a-c)q_1-bq_1^2-bq_1q_2$$
\bsk
Hold $q_2$ fixed. First order:
\bsk
$\dfrac{\partial \pi_1}{\partial q_1}=$ \underline{\hspace{2.2in}} $=0$ \vspace{0.6in}

Second order:
\bsk
$\dfrac{\partial^2 \pi_1}{\partial q_1^2}=$ \underline{\hspace{1.2in}}, negative because
$b>0$, so the flat point is a maximum. \vspace{0.4in}
\end{frame}

\begin{frame}{From First-Order Condition to Best Reply}
Shell's first-order condition holds at Shell's optimum, \textit{whatever} $q_2$ turns out
to be:
$$a-c-2bq_1-bq_2=0$$
\bsk
So solve it for $q_1$: \vspace{0.9in}

\centering
$\displaystyle BR_1(q_2)=\frac{a-c-bq_2}{2b}$
\vfill
\raggedright
The second-order condition was $-2b<0$, so this really is Shell's best, not its worst.
\vspace{-0.2in}
\end{frame}

\begin{frame}{A Best Reply Is a Function}
$$BR_1(q_2)=\frac{a-c-bq_2}{2b}$$
\bsk
Feed it a quantity for Speedway, and it hands back Shell's best answer to that quantity.
One input, one output.
\bsk
\begin{itemize}
    \item If Speedway floods the market, $BR_1$ tells Shell to pull back
    \item If Speedway sits out, $BR_1$ tells Shell to act like a monopolist
\end{itemize}
\bsk
Speedway's problem is the same with the labels swapped:
$$BR_2(q_1)=\frac{a-c-bq_1}{2b}$$
\vfill
{\footnotesize $BR_i(q_{-i})$ is truncated at 0 if needed.}
\vspace{-0.3in}
\end{frame}

\begin{frame}{Nash Equilibrium Is a Fixed Point}
Recall the best-reply version of the definition: $(q_1^*,q_2^*)$ is a Nash equilibrium
exactly when each is a best reply to the other.
$$q_1^*=BR_1(q_2^*) \qquad\text{and}\qquad q_2^*=BR_2(q_1^*)$$
\bsk
Substitute the second into the first:
$$q_1^*=BR_1\big(BR_2(q_1^*)\big)$$
\bsk
Two equations in two unknowns just became \textbf{one} equation in one unknown.
\vfill
Start at $q_1^*$, ask what Speedway would do, then ask what Shell would do back. A Nash
equilibrium is a quantity that returns to \textit{itself} after that round trip.
\vspace{0.3in}
\end{frame}

\begin{frame}{Solving the Fixed Point}
$$q_1=\frac{a-c-b\,BR_2(q_1)}{2b},\qquad BR_2(q_1)=\frac{a-c-bq_1}{2b}$$
\bsk
Substitute, then solve for $q_1$: \vspace{2.4in}

\centering
$\displaystyle q_1^*=\frac{a-c}{3b}$
\end{frame}

\begin{frame}{And Then the Other One}
Feed $q_1^*$ back through Speedway's best reply:
$$q_2^*=BR_2(q_1^*)=\frac{a-c-b\left(\frac{a-c}{3b}\right)}{2b}$$
\bsk
$q_2^*=$ \underline{\hspace{1.6in}} \vspace{0.8in}

\raggedright
No surprise that it matches $q_1^*$: the two stations face identical problems, so the
equilibrium had to be symmetric.
\vfill
Worth checking directly that $BR_1(q_2^*)=q_1^*$. It does.
\vspace{-0.2in}
\end{frame}

\begin{frame}{Cournot: The Numbers}
At $\bar q_1=\bar q_2=\dfrac{a-c}{3b}$:
\bsk
\begin{itemize}
    \item Price: \hspace{0.55in} $\displaystyle P(\bar q_1,\bar q_2)=\frac{a+2c}{3}$
    \bsk
    \item Each profit: \hspace{0.1in} $\displaystyle \frac{(a-c)^2}{9b}$
\end{itemize}
\bsk
Try it: \ $a=25$, \ $b=2$, \ $c=1$.
\bsk
\centering
$\bar q_1=\bar q_2=$ \underline{\hspace{0.8in}} \hspace{0.4in}
$P=$ \underline{\hspace{0.8in}} \hspace{0.4in}
profit $=$ \underline{\hspace{0.8in}}
\vfill
\end{frame}

\begin{frame}{Three Bars on High Street}
Ann, Ben, and Cal each run a bar. Before a home game each independently commits
to a \textbf{Big} promotion or a \textbf{Small} one.
\bsk
\centering
{\small
\begin{matrixgame}[top={Big, Small}, left={Big, Small}, bottom={Cal: Big, Cal: Small}, player names={Ann, Ben, Cal}]
  \payoffmatrix{2,2,2 & 0,3,1 \\ 3,0,1 & 1,1,0}
  \payoffmatrix[pos={5.4,0}]{1,1,3 & 0,0,2 \\ 0,2,0 & 2,2,3}
\end{matrixgame}}
\bsk
\raggedright
{\footnotesize Ann picks the row, Ben the column, Cal the table. Payoffs in order:
Ann, Ben, Cal.}
\vspace{-0.2in}
\end{frame}

\begin{frame}{Three Bars on High Street}
\begin{enumerate}
    \item Does anyone have a dominant strategy? \vspace{0.25in}
    \item Does IDSDS delete anything at all? \vspace{0.25in}
    \item Find every Nash equilibrium. Underline Ann's best payoff in each column,
      Ben's in each row, and Cal's whenever it beats her payoff in the same cell
      of the other table. \vspace{0.35in}
    \item Is either equilibrium Pareto superior to the other? \vspace{0.25in}
\end{enumerate}
\end{frame}

\begin{frame}{What Have We Learned, Charlie Brown?}
\begin{itemize}
    \item \textbf{Second-price auction}: truthful bidding is weakly dominant (Vickrey)
    \item \textbf{Pivotal mechanism}: truthful reporting is weakly dominant (Clarke)
    \item \textbf{IDSDS}: delete strictly dominated strategies, repeat (order-free,
      justified by common knowledge of rationality)
    \item \textbf{IDWDS}: order matters, so delete simultaneously, and the
      justification is murkier
    \item \textbf{Nash equilibrium}: nobody gains by deviating alone
    \item $NE\subseteq IDSDS$, but a game may have no Nash equilibrium at all
\end{itemize}
\vfill
Next: what happens when players can randomize.
\vspace{-0.4in}
\end{frame}
